\begin{frame}[t]{No tested mechanism explains 6.3\,nm; spread is untested}
  \twelveTop\twelvedense
  \begingroup\scriptsize
  \setlength{\tabcolsep}{4.5pt}
  \renewcommand{\arraystretch}{1.05}
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{3.95cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}p{2.75cm}@{}}
    \toprule
    Candidate (model V1, post hoc) & Result (calibrated run: $\Vthcc$ missed by \twelveN{-0.294}\,V) & Verdict \\
    \midrule
    Thickness error, $t =$ 5.3\,nm & \twelveN{+0.036}\,V (12\,\%); via $\dEc$ alone needs $t \approx$ 2.1\,nm
      (67\,\% error) & rejected \\
    Fewer donors & \twelveN{+0.030}\,V, a factor of ten short & rejected \\
    Back-surface charge & matches $\Vthcc$, but the swing is off by \twelveN{-64.1}\,mV/dec & rejected as sole
      cause \\
    Front $D_{\mathrm{it}}$\,$\times$5 & \twelveN{+0.051}\,V; the full miss needs $\times$23--24, at 6.3\,nm
      only & rejected at plausible $D_{\mathrm{it}}$ \\
    Combination & 0.215 of 0.294\,V (73\,\%); swing off by \twelveN{-45.9}\,mV/dec & rejected \\
    No confinement law & misses by only \twelveN{-0.057}\,V; the on-state fails as in every run & not an
      explanation \\
    Film-specific front charge & $Q_f =$ \num{8.7e10} instead of \num{1.73e12}\cmsq{} fits & a calibration
      \lab{fitted} \\
    \midrule
    Uniform donors (6.3--13.2\,nm fit)$^{\ast}$ & \num{2.2e18}\cmcu: \twelveN{-0.25}\,V for 2.0--6.3\,nm (measured
      \twelveN{-0.018}\,V) & rejected \\
    \bottomrule
  \end{tabularx}\par\endgroup
  \src{Rejections are model-conditional (model V1 calibration) \twelveLab{computed}; $^{\ast}$1-D surrogate, not ATLAS.
    Run 0014; C028. Data: Prof.\ M.-H.\ Lee's group, NTU.}
  \vspace{-0.6mm}
  \begin{itemize}\setlength{\itemsep}{0pt}
    \item The miss: a rigid 0.294\,V (flat within 3\,mV, \num{e-11}--\num{e-9}\,A/\textmu m; 0.057\,V without the
      $\dEc$ law) plus a shape part, \twelveN{+0.16}/\twelveN{+0.19}\,V at \num{e-7}/\num{2e-7}\,A/\textmu m, the
      only part independent of the $\dEc$ law and $Q_f$ \lab{computed}.
    \item Untested: device spread\sref{Januar2026}, batch, sweep history, real non-monotonic electrostatics (prior
      low). Decisive: $\ge$10 devices at 2, 4, 6.3, 9, 13\,nm from one run; the last is falsified by a smooth trend.
  \end{itemize}
  \takeaway{For now, spread or a fabrication difference explains 6.3\,nm most economically.}
\end{frame}

\begin{frame}[t]{Confinement can deactivate shallow donors, independent of W}
  \twelvedense
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.60\textwidth}
      \begin{itemize}
        \item Bulk hydrogenic donor, $m^*$ = 0.208\,$m_0$, $\varepsilon_s$ = 10.55\sref[p.~225102-1]{Stokey2021}:
          $E_D$ = 25.4\,meV, $a_B^*$ = 2.68\,nm, Mott density $n_c$ = \num{9.1e17}\cmcu{} \lab{derived}; derivation: backup.
        \item At $t$ = 2\,nm $\approx$ 0.75\,$a_B^*$ confinement roughly doubles $E_D$; the strict 2-D limit is four
          times, 0.10\,eV. Weak screening by the surroundings raises the 2-D bound to 0.45\,eV (\AlO/air) or
          0.11\,eV (\HfO/air, nearer the stack)\srefs{Bastard1981b,Cudazzo2011} \lab{theory}.
        \item 2\,\% W fully active gives \num{6e20}--\num{1.8e21}\cmcu; the device fit needs
          \num{2.5e17}--\num{1.8e18}\cmcu: compensation or inactive donors\sref{Lany2007}, an argument of scale
          \lab{inferred}.
      \end{itemize}
    \end{column}
    \begin{column}{0.37\textwidth}
      \centering
      \includegraphics[width=0.9\linewidth]{12_binding}\par
      \raggedright
      \src{Strict 2-D bound $4\,\mathrm{Ry}\,(m^*/m_0)/\bar\varepsilon^{\,2}$, $\bar\varepsilon = (\varepsilon_1 +
        \varepsilon_2)/2$ (\AlO{} $\approx$ 9, \HfO{} 19.57, air 1); the device stack lies between the points, nearer
        \HfO/air at large distances (review correction V6). Dotted: $\varepsilon_s$; dashed: bulk $E_D$.
        \twelveLab{theory}; FACT-080; fit: C016.}
    \end{column}
  \end{columns}
  \takeaway{A W-independent route to fewer active donors; order-of-magnitude, not computed here.}
\end{frame}
