\begin{frame}{The channels are disordered; Parts 5--10 use crystal DFT}
  \elevendense

\begin{frame}{H-N classifies each W cell by frozen observables}
  \elevendense
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      {\footnotesize Final SCF, bands $n > N_v$, $\elevenNex = n_{\mathrm{W}}$ (number of W):\par}
      \vspace{-1.6mm}
      {\footnotesize
      \begin{align*}
        \elevenfext &= \tfrac{1}{\elevenNex}\textstyle\sum f_{n\mathbf{k}}w_{\mathbf{k}}\,
          \mathbf{1}\!\left[\PRn \ge 0.20,\ \varepsilon_{n\mathbf{k}} \ge \elevenEh\right]\\
        \elevenflocW &= \tfrac{1}{\elevenNex}\textstyle\sum f_{n\mathbf{k}}w_{\mathbf{k}}\,
          \mathbf{1}\!\left[\PRn \le 0.05,\ p_{\mathrm{W5d}} \ge 0.30\right]\\
        \elevenDW &= \min\{\varepsilon_{n\mathbf{k}} : p_{\mathrm{W5d}} \ge 0.30,\ \varepsilon_{n\mathbf{k}} >
          \elevenEhv\} - \elevenEh
      \end{align*}\par}
      \vspace{-1.2mm}
      {\scriptsize $N_v = (N_{\mathrm{el}} - \elevenNex)/2$; $f, w$: occupation, $k$
        weight; $p_{\mathrm{W5d}}$: W-5d weight; $\elevenEh$: host edge, lowest state above $N_v$ with
        $\PRn \ge 0.20$ and interior weight (W weight not used); $\elevenEhv$: the same rule at or below $N_v$.\par}
      \vspace{1.0mm}
      {\footnotesize Participation ratio; L\"owdin weights $p_i$, $N$ atoms\srefs{BellDean1970,Thouless1974}:\par}
      \vspace{-1.6mm}
      {\footnotesize\[ \PRn = \frac{1}{N\sum_i p_i^2}, \qquad \elevenNeff = N\,\PRn \]\par}
      \vspace{-1.2mm}
      {\scriptsize Extended $\PRn \ge 0.20$, localised $\PRn \le 0.05$ (absolute); $\elevenNeff$ always reported.\par}
