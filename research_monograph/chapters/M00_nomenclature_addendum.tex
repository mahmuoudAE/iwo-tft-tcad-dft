\section*{Additional symbols of the monograph}
\begin{longtable}{@{}p{2.6cm}p{11.5cm}@{}}
\toprule
Symbol & Meaning \\\midrule\endhead
$\tHH$ & film thickness between the outermost H atoms of a crystalline slab (the abscissa of every crystalline law) \\
$t_{\mathrm{eq}}$ & mass-equivalent thickness of an amorphous slab (film mass / 7.0~g~cm$^{-3}$ / area) \\
$\Delta E_{\mathrm{v}}$ & signed shift of the valence-band maximum relative to bulk (negative = downwards); $\Delta\Eg = \dEc - \Delta E_{\mathrm{v}}$ \\
$\alpha$, $\alpha_\parallel$ & Kane non-parabolicity of the bulk band and of a confined subband in-plane (eV$^{-1}$) \\
$m_b$, $m_\parallel$ & bulk band-edge mass and in-plane band-edge mass of a confined subband \\
$E_P$ & interband energy of the two-band $k\cdot p$ model \\
$r$ & ratio of HSE06 to PBE gap openings, $\Delta\Eg^{\mathrm{HSE}}/\Delta\Eg^{\mathrm{PBE}}$ \\
$N_{2D}$ & two-dimensional effective density of states of one subband (cm$^{-2}$) \\
$N_V$, $m_{h,\mathrm{DOS}}$ & valence effective density of states and the equivalent hole mass \\
$\PRn$, $N_{\mathrm{eff}}$ & normalised participation ratio of state $n$ and the number of atoms it occupies, $N\PRn$ \\
$N_v$, $N_{\mathrm{ex}}$ & electron-accounting band index and nominal excess-electron count of an amorphous cell \\
$f_{\mathrm{ext}}$, $f_{\mathrm{locW}}$, $\Delta_W$ & H-N observables: extended-state fraction, W-localised fraction, W-level separation from the host edge \\
$\Delta E_{\mathrm{local}}$, $\Delta E_{\mathrm{dop}}$ & H-L vacancy energy differences, near against far from W and doped against undoped \\
$E_D$, $a_B^*$ & hydrogenic donor binding energy and Bohr radius \\
$J$ & equal-film objective of the shared-model campaign \\
$\Delta V_0$, $\mu_\infty$, $\ell$, $N_s$, $W_T$ & shared fitted parameters of campaign C10 \\\bottomrule
\end{longtable}
